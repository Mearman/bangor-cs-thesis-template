\chapter{Design}
Present the design decisions and the reasons for them. An algorithm
float shows the shape of a method; Listing~\ref{lst:sample} shows the
shape of a code listing.

\begin{algorithm}[htbp]
  \caption{A sample algorithm float.}
  \label{alg:sample}
  \begin{algorithmic}[1]
    \State $result \gets 0$
    \ForAll{$item$ in $items$}
      \State $result \gets result + item$
    \EndFor
    \State \Return $result$
  \end{algorithmic}
\end{algorithm}
