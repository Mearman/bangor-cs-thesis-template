\chapter{Design}
Present the design decisions and the reasons for them. \Cref{eq:information-gain} formalises the measure the design optimises.

\begin{equation}
  \label{eq:information-gain}
  \mathrm{IG}(S, a) = \mathrm{H}(S) - \sum_{v \in \mathrm{vals}(a)}
  \frac{|S_v|}{|S|}\,\mathrm{H}(S_v)
\end{equation}

\Cref{alg:sample} shows the shape of an algorithm float. Write the steps with the state, loop, and return commands the algpseudocode package provides.

\begin{algorithm}[htbp]
  \caption{A sample algorithm float.}
  \label{alg:sample}
  \begin{algorithmic}[1]
    \State $result \gets 0$
    \ForAll{$item$ in $items$}
      \State $result \gets result + item$
    \EndFor
    \State \Return $result$
  \end{algorithmic}
\end{algorithm}
