\section{Algorithms}
\Cref{alg:sample} shows the shape of an algorithm float. Write the steps with the state, loop, and return commands the algpseudocode package provides.

\begin{algorithm}[htbp]
  \caption{A sample algorithm float.}
  \label{alg:sample}
  \begin{algorithmic}[1]
    \State $result \gets 0$
    \ForAll{$item$ in $items$}
      \State $result \gets result + item$
    \EndFor
    \State \Return $result$
  \end{algorithmic}
\end{algorithm}
